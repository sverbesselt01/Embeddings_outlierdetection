// Load collection.
var dataset = ee.ImageCollection('GOOGLE/SATELLITE_EMBEDDING/V1/ANNUAL');

// Your AOI
var aoi = geometry;

// Filter by time + region.
var filteredCollection = dataset
  .filterDate('2023-01-01', '2024-01-01')
  .filterBounds(aoi);

// Generate list of all 64 band names A01–A64 automatically.
var bandNames = ee.List.sequence(0, 63)
  .map(function(n) {
    return ee.String('A').cat(ee.Number(n).format('%02d'));
  });


print({ "Bands used:": bandNames });

var scale = 10;

// Count images.
filteredCollection.size().evaluate(function(count) {

  print({ "Number of images": count });

  if (count === 0) {
    print("No images found.");
    return;
  }

  var imageList = filteredCollection.toList(count);

  for (var i = 0; i < count; i++) {

    var image = ee.Image(imageList.get(i));

    // Get server-side ID and convert once
    var id = image.id();
    var idClient = id.getInfo();  // OK to call once per export

    Export.image.toDrive({
      image: image.select(bandNames).clip(aoi),
      description: idClient + '_A00_A63',
      fileNamePrefix: idClient + '_A00_A63',
      folder: 'GEE_Exports',
      region: aoi,
      scale: scale,
      crs: 'EPSG:4326',
      maxPixels: 1e13
    });
  }
});
