// =======================
// Function: cosine similarity
// =======================
function similarityToPoint(pt) {
  // Small buffer (~50 m) for 10 m resolution embeddings
  var vecDict = image.reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: pt.geometry().buffer(50),
    scale: 10,
    maxPixels: 1e9
  });

  // Build numeric array safely, replacing nulls with 0
  var vecArray = ee.Array(
    image.bandNames().map(function(b) {
      var v = ee.Number(vecDict.get(b));
      return ee.Algorithms.If(v, v, 0);
    })
  );

  // Convert array → image (one band per embedding dimension)
  var vecImg = ee.Image.constant(vecArray)
    .arrayFlatten([image.bandNames()]);

  // Dot product → similarity
  return image.multiply(vecImg).reduce(ee.Reducer.sum());
}

// =======================
// Compute similarity maps (3 refs)
// =======================
var sims = refPoints.toList(3).map(function(f) {
  return similarityToPoint(ee.Feature(f));
});

var similarityMean = ee.ImageCollection.fromImages(sims).mean();

// =======================
// Z-score normalization
// =======================
var stats = similarityMean.reduceRegion({
  reducer: ee.Reducer.mean().combine({
    reducer2: ee.Reducer.stdDev(),
    sharedInputs: true
  }),
  geometry: roi,
  scale: 10,
  maxPixels: 1e13,
  tileScale: 4
});

var mean = ee.Number(ee.Algorithms.If(stats.get('sum_mean'), stats.get('sum_mean'), 0));
var std = ee.Number(ee.Algorithms.If(stats.get('sum_stdDev'), stats.get('sum_stdDev'), 1));
std = ee.Number(ee.Algorithms.If(std.eq(0), 1, std));

var zScore = similarityMean.subtract(mean).divide(std);

// =======================
// Thresholding
// =======================
var threshold = 2;
var highSimilarity = zScore.gt(threshold).selfMask();

// =======================
// Visualization
// =======================
Map.addLayer(
  zScore.clip(roi),
  {min: -2, max: 2, palette: ['blue', 'white', 'red']},
  'Z-score cosine similarity'
);

Map.addLayer(
  highSimilarity.clip(roi),
  {palette: ['red']},
  'High similarity areas'
);

Map.centerObject(roi, 8);
Map.setOptions('SATELLITE');
